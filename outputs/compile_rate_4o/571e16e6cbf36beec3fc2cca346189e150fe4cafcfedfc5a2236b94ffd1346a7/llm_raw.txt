bool kissat_restarting(kissat *solver) {
  assert(solver->unassigned);

  if (!GET_OPTION(restart))
    return false;
  if (!solver->level)
    return false;

  uint64_t conflicts = CONFLICTS;
  uint64_t decisions = DECISIONS;
  uint64_t restart_conflicts = solver->limits.restart.conflicts;

  if (conflicts < restart_conflicts)
    return false;

  if (solver->stable) {
    if (kissat_reluctant_triggered(&solver->reluctant))
      return true;
  } else {
    double fast_glue = AVERAGE(fast_glue);
    double slow_glue = AVERAGE(slow_glue);
    double margin = (100.0 + GET_OPTION(restartmargin)) / 100.0;
    double limit = margin * slow_glue;

    if (fast_glue >= limit)
      return true;
  }

  // Micro restart logic
  if (conflicts % 80 == 0) {
    double decision_rate = AVERAGE(decision_rate);
    if (decision_rate < 0.5)  // Example threshold
      return true;
  }

  // Meso restart logic
  if (conflicts % 800 == 0) {
    double reward_variance = fabs(solver->mab_reward[0] - solver->mab_reward[1]);
    if (reward_variance > 0.1)  // Example threshold
      return true;
  }

  // Macro restart logic
  if (conflicts % 8000 == 0) {
    double vsids_score = solver->mab_reward[0] / (solver->mab_select[0] + 1);
    double chb_score = solver->mab_reward[1] / (solver->mab_select[1] + 1);
    double weighted_score = vsids_score * 0.7 + chb_score * 0.3;  // Example weights

    if (weighted_score > 0.6)  // Example threshold
      return true;
  }

  return false;
}
