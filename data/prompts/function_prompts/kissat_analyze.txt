## 1. FUNCTION CONTRACT

```c
int kissat_analyze (kissat *solver, clause *conflict);
```

**Purpose:** Perform conflict analysis: determine the learned clause via first-UIP resolution, minimize/shrink it, bump variable activities, learn the clause, and backtrack. This is the core CDCL conflict analysis routine.

**Parameters:**
| Parameter | Type | Description |
|-----------|------|-------------|
| `solver` | `kissat *` | Solver instance |
| `conflict` | `clause *` | The conflicting clause |

**Return value:**
- `0` — analysis complete, search continues
- `20` — UNSAT (inconsistent at level 0)

**Caller context:** Called from the main search loop in `search.c` when propagation finds a conflict:
```c
clause *conflict = kissat_search_propagate(solver);
if (conflict)
  res = kissat_analyze(solver, conflict);
```

### Analysis Loop Pattern
The standard analysis loop handles multiple cases:
1. **Inconsistent at level 0** → return 20
2. **One literal at conflict level** → reuse conflict as driving clause
3. **Conflict at level 1** → failed literal analysis
4. **General case** → first-UIP deduction → minimize → shrink → bump → learn
5. **Deduction finds new conflict** → restart analysis loop

static bool
one_literal_on_conflict_level (kissat *solver, clause *conflict,
                               unsigned *conflict_level_ptr) {
  /* Checks if conflict has 1 literal at highest level.
     Returns true if conflict reused as driving clause.
     Sets solver->inconsistent if conflict at level 0. */
  assert (conflict);
  assert (conflict->size > 1);
  unsigned conflict_level = INVALID_LEVEL;
  unsigned literals_on_conflict_level = 0;
  unsigned forced_lit = INVALID_LIT;
  assigned *all_assigned = solver->assigned;
  unsigned *lits = conflict->lits;
  const unsigned conflict_size = conflict->size;
  const unsigned *const end_of_lits = lits + conflict_size;
  for (const unsigned *p = lits; p != end_of_lits; p++) {
    const unsigned lit = *p;
    assert (VALUE (lit) < 0);
    const unsigned idx = IDX (lit);
    const unsigned level = all_assigned[idx].level;
    if (conflict_level == level) {
      if (++literals_on_conflict_level > 1 && level == solver->level) break;
    } else if (conflict_level == INVALID_LEVEL || conflict_level < level) {
      forced_lit = lit;
      conflict_level = level;
      literals_on_conflict_level = 1;
    }
  }
  *conflict_level_ptr = conflict_level;
  if (!conflict_level) {
    solver->inconsistent = true;
    CHECK_AND_ADD_EMPTY ();
    ADD_EMPTY_TO_PROOF ();
    return false;
  }
  if (conflict_level < solver->level)
    kissat_backtrack_after_conflict (solver, conflict_level);
  /* ... watch reordering for large clauses ... */
  if (literals_on_conflict_level > 1) return false;
  kissat_backtrack_after_conflict (solver, solver->level - 1);
  if (conflict_size == 2) {
    const unsigned other = lits[0] ^ lits[1] ^ forced_lit;
    kissat_assign_binary (solver, conflict->redundant, forced_lit, other);
  } else {
    const reference ref = kissat_reference_clause (solver, conflict);
    kissat_assign_reference (solver, forced_lit, ref, conflict);
  }
  return true;
}

/* ... analyze_reason_side_literals, sort_deduced_clause,
       reset_levels, kissat_reset_only_analyzed_literals,
       reset_removable, reset_analysis_but_not_analyzed_literals,
       update_trail_average, update_decision_rate_average,
       analyze_failed_literal defined above ... */

int kissat_analyze (kissat *solver, clause *conflict) {
  for (all_literals_in_clause (lit, conflict))
    solver->phases.conflict[IDX(lit)] = lit % 2 ? -1 : 1;
  if (solver->inconsistent) {
    assert (!solver->level);
    return 20;
  }
  START (analyze);
  if (!solver->probing) {
    update_trail_average (solver);
    update_decision_rate_average (solver);
#ifndef QUIET
    UPDATE_AVERAGE (level, solver->level);
#endif
  }
  int res;
  do {
    LOGCLS (conflict, "analyzing conflict %" PRIu64, CONFLICTS);
    unsigned conflict_level;
    if (one_literal_on_conflict_level (solver, conflict, &conflict_level))
      res = 1;
    else if (!conflict_level)
      res = -1;
    else if (conflict_level == 1) {
      analyze_failed_literal (solver, conflict);
      res = 1;
    } else if ((conflict = kissat_deduce_first_uip_clause (solver, conflict))) {
      reset_analysis_but_not_analyzed_literals (solver);
      res = 0;
    } else {
      if (GET_OPTION (minimize)) {
        sort_deduced_clause (solver);
        kissat_minimize_clause (solver);
        if (GET_OPTION (shrink))
          kissat_shrink_clause (solver);
      }
      analyze_reason_side_literals (solver);
      kissat_learn_clause (solver);
      reset_analysis_but_not_analyzed_literals (solver);
      res = 1;
    }
    if (!EMPTY_STACK (solver->analyzed)) {
      if (!solver->probing)
        kissat_bump (solver);
      kissat_reset_only_analyzed_literals (solver);
    }
  } while (!res);
  STOP (analyze);
  return res > 0 ? 0 : 20;
}
