bool kissat_restarting(kissat *solver) {
  assert(solver->unassigned);

  if (!GET_OPTION(restart))
    return false;
  if (!solver->level)
    return false;
  if (CONFLICTS < solver->limits.restart.conflicts)
    return false;

  const double fast = AVERAGE(fast_glue);
  const double slow = AVERAGE(slow_glue);
  const double margin = (100.0 + GET_OPTION(restartmargin)) / 100.0;
  const double limit = margin * slow;

  if (solver->stable)
    return kissat_reluctant_triggered(&solver->reluctant);

  // Dual-threshold learning rate adaptation
  const double improvement = slow - fast;
  const double upper_threshold = 0.1;  // Example threshold
  const double lower_threshold = -0.1; // Example threshold
  const double alpha_low = 0.01;       // Example sensitivity
  const double alpha_high = 0.01;      // Example sensitivity
  const double beta = 0.99;            // Example decay factor

  static double lambda = 1.0;          // Learning rate
  static double alpha_low_decay = alpha_low;
  static double alpha_high_decay = alpha_high;

  if (improvement < lower_threshold) {
    lambda *= (1 + alpha_low_decay * (improvement - lower_threshold));
  } else if (improvement > upper_threshold) {
    lambda *= (1 - alpha_high_decay * (improvement - upper_threshold));
  }

  alpha_low_decay *= beta;
  alpha_high_decay *= beta;

  return (limit <= fast);
}
