bool kissat_restarting(kissat *solver) {
  assert(solver->unassigned);

  if (!GET_OPTION(restart))
    return false;
  if (!solver->level)
    return false;
  if (CONFLICTS < solver->limits.restart.conflicts)
    return false;

  const double alpha = 0.1; // Sensitivity parameter
  const double beta = 0.9;  // Momentum term
  static double lambda = 1.0; // Learning rate, initialized to 1.0
  static double previous_lambda = 1.0; // Previous learning rate

  const double fast = AVERAGE(fast_glue);
  const double slow = AVERAGE(slow_glue);
  const double margin = (100.0 + GET_OPTION(restartmargin)) / 100.0;
  const double limit = margin * slow;

  double improvement = (slow - fast) / slow;
  double threshold = (fast + slow) / 2.0; // Dynamic threshold

  // Update learning rate with momentum
  lambda = beta * previous_lambda + (1 - beta) * (lambda * (1 + alpha * (improvement - threshold)));
  previous_lambda = lambda;

  if (solver->stable)
    return kissat_reluctant_triggered(&solver->reluctant);

  return (limit <= fast);
}
