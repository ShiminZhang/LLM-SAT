bool kissat_restarting(kissat *solver) {
  assert(solver->unassigned);

  if (!GET_OPTION(restart))
    return false;
  if (!solver->level)
    return false;
  if (CONFLICTS < solver->limits.restart.conflicts)
    return false;

  // Exploration mechanism: epsilon-greedy strategy
  double epsilon = 0.1; // Exploration rate
  bool explore = ((double)rand() / RAND_MAX) < epsilon;

  // Calculate context features
  double fast_glue = AVERAGE(fast_glue);
  double slow_glue = AVERAGE(slow_glue);
  double decision_rate = AVERAGE(decision_rate);
  double margin = (100.0 + GET_OPTION(restartmargin)) / 100.0;
  double limit = margin * slow_glue;

  // Nonlinear model: simple neural network with one hidden layer
  double hidden_layer[3];
  hidden_layer[0] = tanh(fast_glue * 0.5 + slow_glue * 0.3 + decision_rate * 0.2);
  hidden_layer[1] = tanh(fast_glue * 0.4 + slow_glue * 0.4 + decision_rate * 0.2);
  hidden_layer[2] = tanh(fast_glue * 0.3 + slow_glue * 0.5 + decision_rate * 0.2);

  double output = hidden_layer[0] * 0.3 + hidden_layer[1] * 0.3 + hidden_layer[2] * 0.4;

  // Dynamic learning rate adjustment based on reward variance
  double reward_variance = fabs(solver->mab_reward[0] - solver->mab_reward[1]);
  double learning_rate = 0.01 / (1.0 + reward_variance);

  // Decision based on exploration or model output
  if (explore) {
    return kissat_reluctant_triggered(&solver->reluctant);
  } else {
    return (output > 0.5) && (limit <= fast_glue);
  }
}
