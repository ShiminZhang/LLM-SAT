bool kissat_restarting(kissat *solver) {
  assert(solver->unassigned);

  if (!GET_OPTION(restart))
    return false;
  if (!solver->level)
    return false;
  if (CONFLICTS < solver->limits.restart.conflicts)
    return false;

  static double momentum_index = 1.0;
  static double stability_factor = 0.1;
  static double past_ratios[5] = {0};
  static int ratio_index = 0;

  double current_ratio = (double)solver->mab_conflicts / solver->mab_decisions;
  past_ratios[ratio_index] = current_ratio;
  ratio_index = (ratio_index + 1) % 5;

  double average_ratio = 0.0;
  for (int i = 0; i < 5; i++) {
    average_ratio += past_ratios[i];
  }
  average_ratio /= 5.0;

  double variance = 0.0;
  for (int i = 0; i < 5; i++) {
    double diff = past_ratios[i] - average_ratio;
    variance += diff * diff;
  }
  variance /= 5.0;

  if (current_ratio > average_ratio) {
    momentum_index += 0.05 + stability_factor;
  } else {
    momentum_index -= 0.05;
  }

  if (variance < 0.1) {
    stability_factor += 0.01;
  } else if (variance > 0.2) {
    stability_factor -= 0.01;
  }

  if (stability_factor < 0.05) stability_factor = 0.05;
  if (stability_factor > 0.2) stability_factor = 0.2;

  if (momentum_index < 0.5) momentum_index = 0.5;
  if (momentum_index > 1.5) momentum_index = 1.5;

  uint64_t base_restart_interval = GET_OPTION(restartint);
  uint64_t adjusted_restart_interval = (uint64_t)(base_restart_interval * momentum_index);

  if (CONFLICTS >= adjusted_restart_interval) {
    return true;
  }

  return false;
}
