bool kissat_restarting(kissat *solver) {
  assert(solver->unassigned);

  if (!GET_OPTION(restart))
    return false;
  if (!solver->level)
    return false;

  uint64_t conflicts = CONFLICTS;
  uint64_t nano_threshold = 50;
  uint64_t micro_threshold = 200;
  uint64_t meso_threshold = 1500;
  uint64_t macro_threshold = 12000;

  if (conflicts < solver->limits.restart.conflicts)
    return false;

  if (conflicts < nano_threshold)
    return false;

  if (conflicts < micro_threshold) {
    double decision_rate = AVERAGE(decision_rate);
    if (decision_rate > 0.5)  // Example condition for micro restart
      return true;
  }

  if (conflicts < meso_threshold) {
    double fast_glue = AVERAGE(fast_glue);
    double slow_glue = AVERAGE(slow_glue);
    double margin = (100.0 + GET_OPTION(restartmargin)) / 100.0;
    double limit = margin * slow_glue;
    if (fast_glue > limit)
      return true;
  }

  if (conflicts < macro_threshold) {
    if (solver->stable)
      return kissat_reluctant_triggered(&solver->reluctant);
    else {
      double reward_ratio = solver->mab_reward[solver->heuristic] /
                            (solver->mab_select[solver->heuristic] + 1);
      if (reward_ratio < 0.1)  // Example condition for macro restart
        return true;
    }
  }

  return false;
}
