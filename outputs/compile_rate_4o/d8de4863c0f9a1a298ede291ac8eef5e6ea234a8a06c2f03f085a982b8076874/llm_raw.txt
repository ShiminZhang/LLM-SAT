bool kissat_restarting(kissat *solver) {
  assert(solver->unassigned);

  if (!GET_OPTION(restart))
    return false;
  if (!solver->level)
    return false;

  uint64_t conflicts = CONFLICTS;
  uint64_t nano_threshold = 50;
  uint64_t meso_threshold = 800;
  uint64_t macro_threshold = 5000;

  if (conflicts < solver->limits.restart.conflicts)
    return false;

  if (conflicts < nano_threshold) {
    return false;
  } else if (conflicts < meso_threshold) {
    double fast_glue = AVERAGE(fast_glue);
    double slow_glue = AVERAGE(slow_glue);
    double margin = (100.0 + GET_OPTION(restartmargin)) / 100.0;
    double limit = margin * slow_glue;
    return (limit <= fast_glue);
  } else if (conflicts < macro_threshold) {
    double decision_rate = AVERAGE(decision_rate);
    double reward_to_conflict_ratio = solver->mab_reward[solver->heuristic] / (double)solver->mab_conflicts;
    return (reward_to_conflict_ratio > decision_rate);
  } else {
    double decay_factor = exp(-solver->mab_chosen_tot / (double)solver->mab_decisions);
    double adaptive_threshold = macro_threshold * decay_factor;
    return (conflicts >= adaptive_threshold);
  }
}
