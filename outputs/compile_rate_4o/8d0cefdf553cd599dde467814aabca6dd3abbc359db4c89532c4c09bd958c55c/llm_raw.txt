bool kissat_restarting(kissat *solver) {
  assert(solver->unassigned);

  if (!GET_OPTION(restart))
    return false;
  if (!solver->level)
    return false;
  if (CONFLICTS < solver->limits.restart.conflicts)
    return false;

  static double threshold = 1.0; // Initial threshold
  const double learning_rate = 0.01; // Learning rate for threshold adjustment

  // Calculate Complexity Score (CS)
  double cs = (double)(CONFLICTS * solver->statistics.restarts) / (DECISIONS + 1);

  // Determine if restart should be triggered
  bool restart = false;
  if (cs > threshold * 1.1) { // CS significantly above threshold
    solver->limits.restart.conflicts *= 1.25; // Increase horizon
    threshold += learning_rate; // Increase threshold
    restart = true;
  } else if (cs < threshold * 0.9) { // CS significantly below threshold
    solver->limits.restart.conflicts *= 0.75; // Decrease horizon
    threshold -= learning_rate; // Decrease threshold
  }

  // Optional verbose logging for debugging
  kissat_extremely_verbose(solver, "CS: %f, Threshold: %f, Restart: %s",
                           cs, threshold, restart ? "true" : "false");

  return restart;
}
