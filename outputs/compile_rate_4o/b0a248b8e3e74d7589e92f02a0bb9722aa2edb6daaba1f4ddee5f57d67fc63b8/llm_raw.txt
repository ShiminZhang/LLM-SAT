bool kissat_restarting(kissat *solver) {
  assert(solver->unassigned);

  if (!GET_OPTION(restart))
    return false;
  if (!solver->level)
    return false;
  if (CONFLICTS < solver->limits.restart.conflicts)
    return false;

  const double fast = AVERAGE(fast_glue);
  const double slow = AVERAGE(slow_glue);
  const double margin = (100.0 + GET_OPTION(restartmargin)) / 100.0;
  const double limit = margin * slow;

  if (solver->stable)
    return kissat_reluctant_triggered(&solver->reluctant);

  // Adaptive exploration mechanism
  const unsigned window_size = 30;
  const double min_exploration = 0.5;
  const double max_exploration = 2.0;
  const double exploration_adjustment = 0.15;
  const double threshold = 0.07;

  double current_crr = (double)solver->mab_conflicts / DECISIONS;
  double window_average_crr = 0.0; // Assume a function to calculate this

  // Calculate window average CRR (pseudo-code)
  // window_average_crr = calculate_window_average_crr(solver, window_size);

  double exploration_coefficient = 1.0; // Assume initial value

  if (current_crr > window_average_crr * (1.0 + threshold)) {
    exploration_coefficient *= (1.0 + exploration_adjustment);
  } else if (current_crr < window_average_crr * (1.0 - threshold)) {
    exploration_coefficient *= (1.0 - exploration_adjustment);
  }

  // Clamp exploration coefficient
  if (exploration_coefficient < min_exploration)
    exploration_coefficient = min_exploration;
  if (exploration_coefficient > max_exploration)
    exploration_coefficient = max_exploration;

  // Modify restart interval based on exploration coefficient
  solver->limits.restart.conflicts *= exploration_coefficient;

  return (limit <= fast);
}
