static inline void init_activityref (kissat *solver, activityref *ar,
                                     clause *c, reference ref,
                                     heap *scores) {
  assert (COUNTREF_COUNTS <= c->size);
  assert (c->size <= MAX_COUNTREF_SIZE);
  ar->size = c->size;
  ar->vivify = c->vivify;
  ar->ref = ref;
  double best = DBL_MAX;
  for (all_literals_in_clause (lit, c)) {
    unsigned idx = IDX (lit);
    double activity = kissat_get_heap_score (scores, idx);
    if (activity < best)
      best = activity;
  }
  ar->activity = best;
}

static void init_activityrefs (vivifier *vivifier) {
  kissat *const solver = vivifier->solver;
  references *schedule = &vivifier->schedule;
  activityrefs *activityrefs = &vivifier->activityrefs;
  assert (EMPTY_STACK (*activityrefs));
  assert (solver->stable);
  heap *scores = &solver->scores;
  for (all_stack (reference, ref, *schedule)) {
    clause *c = kissat_dereference_clause (solver, ref);
    activityref activityref;
    init_activityref (solver, &activityref, c, ref, scores);
    PUSH_STACK (*activityrefs, activityref);
  }
  RELEASE_STACK (*schedule);
}

/* Sorting: activityrefs are sorted by ar->activity ascending */
static bool smaller_activity (activityref a, activityref b) {
  return a.activity < b.activity;
}

static void rank_vivification_candidates_by_activity (vivifier *vivifier) {
  kissat *solver = vivifier->solver;
  activityrefs *activityrefs = &vivifier->activityrefs;
  SORT_STACK (activityref, *activityrefs, smaller_activity);
}
