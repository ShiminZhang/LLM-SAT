bool kissat_restarting(kissat *solver) {
  assert(solver->unassigned);

  if (!GET_OPTION(restart))
    return false;
  if (!solver->level)
    return false;

  uint64_t conflicts = CONFLICTS;
  uint64_t nano_threshold = 50;
  uint64_t meso_threshold = 800;
  uint64_t macro_threshold = 5000;

  if (conflicts < solver->limits.restart.conflicts)
    return false;

  if (conflicts < nano_threshold) {
    return false;
  } else if (conflicts < meso_threshold) {
    double reward_conflict_ratio = solver->mab_reward[solver->heuristic] / (double) solver->mab_conflicts;
    double decision_priority = AVERAGE(decision_rate) * reward_conflict_ratio;
    return decision_priority > 1.0;
  } else if (conflicts < macro_threshold) {
    double decay_factor = exp(-solver->mab_chosen_tot / (double) solver->mab_decisions);
    double historical_analysis = AVERAGE(fast_glue) * decay_factor;
    return historical_analysis < AVERAGE(slow_glue);
  } else {
    solver->limits.restart.conflicts = conflicts + GET_OPTION(restartint);
    return true;
  }
}
