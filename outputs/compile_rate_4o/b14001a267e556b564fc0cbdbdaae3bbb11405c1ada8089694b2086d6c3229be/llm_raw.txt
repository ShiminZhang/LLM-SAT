bool kissat_restarting(kissat *solver) {
  assert(solver->unassigned);

  if (!GET_OPTION(restart))
    return false;
  if (!solver->level)
    return false;
  if (CONFLICTS < solver->limits.restart.conflicts)
    return false;

  static double momentum_index = 1.0;
  const double min_momentum = 0.5;
  const double max_momentum = 1.5;
  const double momentum_step = 0.05;

  double current_ratio = (double) solver->mab_conflicts / solver->mab_decisions;
  static double past_ratios[5] = {0};
  static int ratio_index = 0;
  double average_ratio = 0.0;

  for (int i = 0; i < 5; i++) {
    average_ratio += past_ratios[i];
  }
  average_ratio /= 5.0;

  if (current_ratio > average_ratio) {
    momentum_index += momentum_step;
  } else {
    momentum_index -= momentum_step;
  }

  if (momentum_index < min_momentum) {
    momentum_index = min_momentum;
  } else if (momentum_index > max_momentum) {
    momentum_index = max_momentum;
  }

  past_ratios[ratio_index] = current_ratio;
  ratio_index = (ratio_index + 1) % 5;

  uint64_t base_restart_interval = GET_OPTION(restartint);
  uint64_t adjusted_restart_interval = (uint64_t)(base_restart_interval * momentum_index);

  if (CONFLICTS >= solver->limits.restart.conflicts + adjusted_restart_interval) {
    return true;
  }

  if (solver->stable) {
    return kissat_reluctant_triggered(&solver->reluctant);
  }

  const double fast = AVERAGE(fast_glue);
  const double slow = AVERAGE(slow_glue);
  const double margin = (100.0 + GET_OPTION(restartmargin)) / 100.0;
  const double limit = margin * slow;
  return (limit <= fast);
}
